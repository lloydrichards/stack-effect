import { Effect, Layer, Stream } from "effect";
import { LanguageModel, type Response } from "effect/ai";

export const FastModelLive = Layer.effect(LanguageModel.LanguageModel, LanguageModel.make({
  generateText: () => Effect.die("Expected streaming generation"),
  streamText: ({ prompt }) => {
    const completed = prompt.content.filter((message) => message.role === "tool").length;
    if (completed >= 12) return Stream.die("Generation exceeded the model round limit");
    return Stream.fromIterable<Response.StreamPartEncoded>([
      { type: "tool-call", id: String(completed), name: "think", params: { thought: "continue" } },
      { type: "finish", reason: "tool-calls", usage: { inputTokens: completed === 0 ? {} : { total: 1 }, outputTokens: { total: 1 } } },
    ]);
  },
}));
export const SmartModelLive = FastModelLive;
