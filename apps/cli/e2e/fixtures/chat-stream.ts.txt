import assert from "node:assert/strict";
import { Cause, Deferred, Effect, Fiber, Layer, Option, Queue, Schema, Stream } from "effect";
import { LanguageModel, Tool, Toolkit, type Response } from "effect/ai";
import { AiChatService, AiChatServiceLive } from "@repo/ai";
import { FastModelLive } from "./packages/ai/src/LanguageModel";
import { FetchHttpClient, HttpRouter } from "effect/http";
import { RpcSerialization, RpcServer } from "effect/rpc";
import { ChatRpc } from "@repo/domain/ChatRpc";
import { init, subscriptions, update, type Model } from "./apps/client-foldkit-app/src/features/chat";
import { ChatId, type ChatStreamPart } from "@repo/domain/Chat";

// Serve the supplied chat parts over the same RPC transport used by the generated client.
const receive = (parts: ReadonlyArray<ChatStreamPart>) => Effect.scoped(Effect.gen(function* () {
  const chatId = ChatId.make("test-chat");
  const server = RpcServer.layerHttp({ group: ChatRpc, path: "/chat-rpc", protocol: "http" }).pipe(
    Layer.provide(ChatRpc.toLayer(ChatRpc.of({
      chat_start: () => Effect.succeed({ chatId }),
      chat_ask: () => Stream.fromIterable(parts),
    }))),
    Layer.provide(RpcSerialization.layerNdjson),
  );
  const web = yield* Effect.acquireRelease(
    Effect.sync(() => HttpRouter.toWebHandler(server, { disableLogger: true })),
    (web) => Effect.promise(() => web.dispose()),
  );
  const model = { ...init()[0], chatState: "streaming" as const, chatStreaming: true, chatId: Option.some(chatId) };
  const subscription = subscriptions.chatStream;
  return yield* subscription.dependenciesToStream(subscription.modelToDependencies(model)).pipe(
    Stream.runFold((): Model => model,
      (state, message) => update(state, message)[0]),
    Effect.provideService(FetchHttpClient.Fetch, (input, init) => web.handler(new Request(input, init))),
  );
}));

const toolCall = (id: string, name = "lookup"): Response.StreamPartEncoded => ({
  type: "tool-call", id, name, params: {},
});
const finish = (reason: Response.FinishReason, tokens?: number): Response.StreamPartEncoded => ({
  type: "finish", reason,
  usage: { inputTokens: tokens === undefined ? {} : { total: tokens }, outputTokens: { total: 1 } },
});

const tools = Toolkit.make(
  Tool.make("lookup", { success: Schema.String }),
  Tool.make("confirm", { success: Schema.String, needsApproval: true }),
);
const run = (round: (n: number) => ReadonlyArray<Response.StreamPartEncoded>) => {
  let requests = 0;
  const provider = Layer.effect(LanguageModel.LanguageModel, LanguageModel.make({
    generateText: () => Effect.die("Expected streaming generation"),
    streamText: ({ prompt }) => {
      assert(requests < 12, "generation exceeded the model round limit");
      const systems = prompt.content.filter((message) => message.role === "system");
      assert.equal(systems.flatMap((message) => message.content.match(/Answer in French\./g) ?? []).length, 1);
      return Stream.fromIterable(round(++requests));
    },
  }));
  return Effect.gen(function* () {
    const toolkit = yield* tools.pipe(Effect.provide(tools.toLayer({
      lookup: (_, context) => Effect.gen(function* () {
        yield* context.preliminary("progress");
        return "found";
      }),
      confirm: () => Effect.die("An unapproved tool ran"),
    })));
    const service = yield* AiChatService;
    return yield* service.stream({
      prompt: [{ role: "system", content: "Answer in French." }, { role: "user", content: [{ type: "text", text: "Hello" }] }],
      toolkit,
    }).pipe(Stream.runCollect);
  }).pipe(Effect.provide(AiChatServiceLive), Effect.provide(provider));
};

await Effect.runPromise(Effect.gen(function* () {
  const completed = yield* run((n) => n === 1 ? [
    { type: "tool-params-start", id: "a", name: "lookup" },
    { type: "tool-params-delta", id: "a", delta: "{}" },
    { type: "tool-params-end", id: "a" },
    toolCall("a"), finish("tool-calls", 3),
  ] : [
    { type: "reasoning-start", id: "summary" },
    { type: "reasoning-delta", id: "summary", delta: "Checked the result." },
    { type: "reasoning-end", id: "summary" },
    finish("stop", 7),
  ]);
  assert.deepEqual(completed, [
    { _tag: "tool-start", id: "a", name: "lookup", input: "{}" },
    { _tag: "tool-success", id: "a", name: "lookup", output: "found" },
    { _tag: "reasoning", delta: "Checked the result." },
    { _tag: "finish", reason: "stop", usage: { promptTokens: 10, completionTokens: 2, totalTokens: 12 } },
  ]);

  const released = yield* Deferred.make<void>();
  const streamingProvider = Layer.effect(LanguageModel.LanguageModel, LanguageModel.make({
    generateText: () => Effect.die("Expected streaming generation"),
    streamText: () => Stream.make<Response.StreamPartEncoded>(
      { type: "text-start", id: "a" }, { type: "text-delta", id: "a", delta: "first" },
    ).pipe(Stream.concat(Stream.never), Stream.ensuring(Deferred.succeed(released, undefined))),
  }));
  yield* Effect.gen(function* () {
    const ai = yield* AiChatService;
    const first = yield* ai.stream({ prompt: "Hello" }).pipe(Stream.take(1), Stream.runCollect);
    assert.deepEqual(first, [{ _tag: "text", delta: "first" }]);
    assert(yield* Deferred.isDone(released), "ending the consumer must release the provider");

    const opened = yield* Deferred.make<Queue.Dequeue<ChatStreamPart, Cause.Done>>();
    const owner = yield* Effect.gen(function* () {
      yield* Deferred.succeed(opened, yield* ai.chat("Hello"));
      yield* Effect.never;
    }).pipe(Effect.forkChild);
    const queue = yield* Deferred.await(opened);
    assert.deepEqual(yield* Queue.take(queue), { _tag: "text", delta: "first" });
    yield* Fiber.interrupt(owner);
    assert.deepEqual(yield* Queue.collect(queue), []);
  }).pipe(Effect.provide(AiChatServiceLive), Effect.provide(streamingProvider));

  const capped = yield* AiChatService.use((ai) => ai.stream({ prompt: "continue" }).pipe(Stream.runCollect)).pipe(
    Effect.provide(AiChatServiceLive), Effect.provide(FastModelLive),
  );
  assert.equal(capped.filter((part) => part._tag === "tool-success").length, 12);
  assert.deepEqual(capped.filter((part) => part._tag === "finish"), [
    { _tag: "finish", reason: "max-iterations", usage: undefined },
  ]);
  const cappedClient = yield* receive(capped);
  assert.equal(cappedClient.chatState, "complete");
  assert(cappedClient.reachedStepLimit);
  assert.equal(cappedClient.chatHistory.length, 1);
  assert.equal(cappedClient.chatHistory[0]?.role, "assistant");

  const legacy = yield* receive([{ _tag: "reasoning", delta: "Reached maximum iterations (12). Stopping here." }]);
  assert.equal(legacy.chatState, "error");
  assert.equal(legacy.chatStreaming, false);
  assert.match(Option.getOrThrow(legacy.chatError), /ended before/);

  const approval = yield* run(() => [toolCall("a", "confirm"), finish("tool-calls")]);
  const failedClient = yield* receive(approval);
  assert.equal(failedClient.chatState, "error");
  assert.match(Option.getOrThrow(failedClient.chatError), /approval/);
  assert.equal(failedClient.chatHistory.length, 0);
}));
